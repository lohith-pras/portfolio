  <action>
Create src/lib/gsap.ts as the ONLY place in the project that imports from 'gsap'. All other files must import from '@/lib/gsap' to guarantee plugins are registered before first use.

```ts
'use client'

import { gsap } from 'gsap'
import { ScrollTrigger } from 'gsap/ScrollTrigger'
import { DrawSVGPlugin } from 'gsap/DrawSVGPlugin'
import { ScrambleTextPlugin } from 'gsap/ScrambleTextPlugin'

// Register all plugins ONCE at module level — not inside hooks.
// Registering inside hooks re-registers every render (PITFALL).
gsap.registerPlugin(ScrollTrigger, DrawSVGPlugin, ScrambleTextPlugin)

export { gsap, ScrollTrigger, DrawSVGPlugin, ScrambleTextPlugin }
```

CRITICAL rules (from PITFALLS.md):
- 'use client' is required — GSAP touches window, no SSR
- Plugins registered at module top-level, never inside useEffect or useGSAP
- No component imports directly from 'gsap' — always from '@/lib/gsap'
- Modular imports only (gsap/ScrollTrigger, NOT gsap/all)

Update src/app/layout.tsx to load Space Mono and Plus Jakarta Sans via next/font/google.

CRITICAL rules:
- next/font/google calls MUST be at module top-level, never inside a component or conditional
- Use `variable` mode to expose fonts as CSS custom properties
- latin subset covers German umlauts (ä ö ü ß) — no need for latin-ext

```tsx
import type { Metadata } from 'next'
import { Space_Mono, Plus_Jakarta_Sans } from 'next/font/google'
import './globals.css'

const spaceMono = Space_Mono({
  weight: ['400', '700'],
  subsets: ['latin'],
  variable: '--font-space-mono',
  display: 'swap',
})

const plusJakartaSans = Plus_Jakarta_Sans({
  weight: ['300', '400', '500', '600', '700'],
  subsets: ['latin'],
  variable: '--font-plus-jakarta',
  display: 'swap',
})

export const metadata: Metadata = {
  title: 'Lohith Tarikere Prasanna',
  description: 'Builder with intent. AI, wireless, and mobility.',
}

export default function RootLayout({ children }: { children: React.ReactNode }) {
  return (
    <html
      lang="en"
      className={`${spaceMono.variable} ${plusJakartaSans.variable}`}
    >
      <body className="bg-[#0A0A0A] text-[#F0F0F0] antialiased">
        {children}
      </body>
    </html>
  )
}
```

Then update tailwind.config.ts fontFamily to reference the CSS vars:
```ts
fontFamily: {
  display: ['var(--font-space-mono)', 'monospace'],
  body: ['var(--font-plus-jakarta)', 'sans-serif'],
  life: ['var(--font-courier-prime)', 'monospace'],
},
```
